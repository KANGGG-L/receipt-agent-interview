   759	function requestAuthLogout() {
   760	    if (!AuthState.enabled) return; // demo 免密模式无生产注销语义
   761	    if (_authLogoutInFlight) return;
   762	    showCustomConfirmModal({
   763	        title: '退出登录',
   764	        message: '确定退出当前账号吗？退出后需要重新登录才能继续操作。',
   765	        confirmText: '退出登录',
   766	        cancelText: '取消',
   767	        onConfirm: function () { performAuthLogout(); },
   768	        onCancel: function () { /* 取消：保持会话，不假退出 */ }
   769	    });
   770	}

   772	function performAuthLogout() {
   773	    if (_authLogoutInFlight) return;
   774	    _authLogoutInFlight = true;
   775	    syncAuthChrome();
   776	    const btn = document.getElementById('authLogoutBtn');
   777	    if (btn) btn.disabled = true;
   778	
   779	    apiFetch('/api/auth/logout', {
   780	        method: 'POST',
   781	        headers: { 'Content-Type': 'application/json', 'X-Requested-With': 'fetch' },
   782	        credentials: 'same-origin',
   783	        body: '{}'
   784	    })
   785	    .then(function (res) {
   786	        return Promise.all([res.status, res.json().catch(function () { return null; })]);
   787	    })
   788	    .then(function (pair) {
   789	        const httpStatus = pair[0];
   790	        const body = pair[1];
   791	        _authLogoutInFlight = false;
   792	        if (httpStatus >= 200 && httpStatus < 300 && body && body.status === 'success') {
   793	            clearClientAuthCredentials();
   794	            AuthState.account = null;
   795	            syncAuthChrome();
   796	            showLoginPanel('已退出登录');
   797	            if (typeof showToast === 'function') showToast('已退出登录', 'info');
   798	            return;
   799	        }
   800	        // 非成功：不清除本地会话态，避免假退出
   801	        syncAuthChrome();
   802	        const msg = (body && body.msg) ? body.msg : ('退出失败（HTTP ' + httpStatus + '）');
   803	        if (typeof showToast === 'function') showToast(msg, 'error');
   804	    })
   805	    .catch(function (err) {
   806	        _authLogoutInFlight = false;
   807	        syncAuthChrome();
   808	        console.error('退出登录请求异常', err);
   809	        if (typeof showToast === 'function') {
   810	            showToast('网络异常，未能退出。请检查网络后重试。', 'error');
   811	        }
   812	    });
   813	}

    62	    - 只吊销当前请求携带的令牌（Bearer 或 Cookie）并 delete_cookie；
    63	    - 不签发新令牌、不改账号、不扩大任何角色写权限；

    66	    tok, _origin = auth._request_token(request)
    67	    full = auth.parse_token_full(tok) if tok else None
    68	    if full:
