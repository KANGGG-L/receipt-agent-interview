function isExportBusinessIdentityHeader(h) {
    if (!h) return false;
    return h.includes('品名') || h.includes('名称') || h.includes('说明')
        || h.includes('备注') || h.includes('配方') || h.includes('建议')
        || h.includes('供应商') || h.includes('往来')
        || h === '部门' || h.includes('部门名称');
}
